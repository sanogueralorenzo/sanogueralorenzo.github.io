cmake_minimum_required(VERSION 3.22)
project(Amarchy C CXX)
# Compile the already-proven bridge in place; Amarchy owns only its JNI entry points.
set(STEAM_DISPLAY "${CMAKE_CURRENT_SOURCE_DIR}/../../androidsteam/native/display")
add_subdirectory("${STEAM_DISPLAY}" steam-display)
get_target_property(DISPLAY_SOURCES deck-display SOURCES)
list(FILTER DISPLAY_SOURCES EXCLUDE REGEX "src/jni\\.c$")
set(AMARCHY_SOURCES "")
foreach(SOURCE ${DISPLAY_SOURCES})
    if(IS_ABSOLUTE "${SOURCE}")
        list(APPEND AMARCHY_SOURCES "${SOURCE}")
    else()
        list(APPEND AMARCHY_SOURCES "${STEAM_DISPLAY}/${SOURCE}")
    endif()
endforeach()
add_library(amarchy-display SHARED ${AMARCHY_SOURCES} jni.c)
target_include_directories(amarchy-display PRIVATE "${STEAM_DISPLAY}/src" "${CMAKE_CURRENT_BINARY_DIR}/steam-display")
target_compile_definitions(amarchy-display PRIVATE _GNU_SOURCE)
target_compile_options(amarchy-display PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(amarchy-display PRIVATE wayland adrenotools android log dl)
