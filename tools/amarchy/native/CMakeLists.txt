cmake_minimum_required(VERSION 3.22)
project(Amarchy C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DECK_DEPS)
    message(FATAL_ERROR "DECK_DEPS must point to verified Wayland/libffi sources")
endif()
set(WAYLAND "${DECK_DEPS}/wayland")
set(GENERATED "${DECK_DEPS}/generated")
add_library(wayland STATIC
    ${WAYLAND}/src/wayland-server.c ${WAYLAND}/src/wayland-shm.c
    ${WAYLAND}/src/event-loop.c ${WAYLAND}/src/connection.c
    ${WAYLAND}/src/wayland-os.c ${WAYLAND}/src/wayland-util.c ${GENERATED}/wayland-protocol.c)
target_include_directories(wayland PUBLIC ${WAYLAND}/src ${GENERATED}
    PRIVATE ${WAYLAND} ${DECK_DEPS}/ffi-install/include)
target_compile_definitions(wayland PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(wayland PUBLIC ${DECK_DEPS}/ffi-install/lib/libffi.a m)
set(KEYMAP_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../../androidsteam/native/display/input/us.xkb")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${KEYMAP_FILE}")
file(READ "${KEYMAP_FILE}" KEYMAP)
string(REPLACE "\\" "\\\\" KEYMAP "${KEYMAP}")
string(REPLACE "\"" "\\\"" KEYMAP "${KEYMAP}")
string(REPLACE "\n" "\\n\"\n\"" KEYMAP "${KEYMAP}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/keymap.h" "static const char amarchy_keymap[] = \"${KEYMAP}\";\n")
add_library(amarchy-display SHARED jni.c display/input.c display/server.c display/surface.c display/shell.c
    display/presentation.c display/dmabuf.c ${GENERATED}/xdg-shell-protocol.c ${GENERATED}/linux-dmabuf-protocol.c)
target_include_directories(amarchy-display PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(amarchy-display PRIVATE _GNU_SOURCE)
target_compile_options(amarchy-display PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(amarchy-display PRIVATE wayland android log)
